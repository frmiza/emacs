#+begin_src emacs-lisp
   ;; ====================================================================
   ;; 3. CONFIGURAÇÃO DO EGLOT (MOTOR LSP NATIVO) + PYRIGHT
   ;; ====================================================================
   (use-package eglot
     :ensure nil ;; Nativo no Emacs 29+
     :defer t
     :config
     ;; CRUCIAL: Desativa logs do Eglot. Se deixar ativado, o Emacs escreve milhares 
     ;; de linhas de log em background a cada movimento do seu cursor, travando a UI.
     (setq eglot-events-buffer-size 0)
     
     ;; Conecta de forma assíncrona para não congelar o editor ao abrir o arquivo
     (setq eglot-sync-connect 0)
     
     ;; Vincula o executável do Pyright ao modo Tree-Sitter do Python
     (add-to-list 'eglot-server-programs
                  '(python-ts-mode . ("pyright-langserver" "--stdio")))

     ;; OTIMIZAÇÃO DE BACKEND: Quando o LSP estiver ativo, força o Company a olhar
     ;; APENAS para os dados do Pyright (capf), ignorando buscas de texto puro.
     (add-hook 'eglot-managed-mode-hook
               (lambda ()
                 (setq-local company-backends '(company-capf)))))

  (use-package python
    :mode ("\\.py\\'" . python-ts-mode)
   :bind (:map python-ts-mode-map
               ("<f5>" . recompile)
               ("<f6>" . eglot-format))
   :hook ((python-ts-mode . eglot-ensure))) 
#+end_src
